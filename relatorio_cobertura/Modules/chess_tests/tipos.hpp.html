constexpr int diagonal_index(int sq) {
    int rank = sq / 8;
    int file = sq % 8;

}
// Retorna o Ã­ndice da antidiagonal (0 a 14) contendo a casa sq (0 a 63)
constexpr int antidiagonal_index(int sq) {
    int rank = sq / 8;
    int file = sq % 8;
    return (rank + file);
}
